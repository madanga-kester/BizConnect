const LIMIT = 20;
const pageRequest = (cursor?: string) => ({ cursor, limit: LIMIT });
const errorText = (error: { message: string; code?: string }) =>
  `${error.message}${error.code ? ` (${error.code})` : ""}`;

export function ApiHomeFeed() {
  const services = useMabizServices();
  const app = useAppStore();
  const [mode, setMode] = useState<"for-you" | "following">("for-you");
  const [posts, setPosts] = useState<Post[]>([]);
  const [cursor, setCursor] = useState<string | undefined>();
  const [loading, setLoading] = useState(true);
  const [loadingMore, setLoadingMore] = useState(false);
  const [refreshing, setRefreshing] = useState(false);
  const [error, setError] = useState("");
  const [quickMenu, setQuickMenu] = useState(false);
  const hasUnread = app.notifications.some((item) => !item.read);
  const load = useCallback(
    async (refresh = false, next = false) => {
      if (next && (!cursor || loadingMore)) return;
      if (refresh) setRefreshing(true);
      else if (next) setLoadingMore(true);
      else setLoading(true);
      setError("");
      const request: FeedFilters = { mode, limit: LIMIT, ...(next && cursor ? { cursor } : {}) };
      const result = await services.feed.list(request);
      if (!result.ok) setError(errorText(result.error));
      else {
        setPosts((current) => (refresh || !next ? result.data.items : [...current, ...result.data.items]));
        setCursor(result.ok ? result.data.nextCursor : undefined);
      }
      setLoading(false);
      setRefreshing(false);
      setLoadingMore(false);
    },
    [cursor, loadingMore, mode, services]
  );
  useEffect(() => {
    setPosts([]);
    setCursor(undefined);
    void load(false, false);
  }, [mode]);
  const onScroll = (event: {
    nativeEvent: {
      layoutMeasurement: { height: number };
      contentOffset: { y: number };
      contentSize: { height: number };
    };
  }) => {
    const { layoutMeasurement, contentOffset, contentSize } = event.nativeEvent;
    if (contentSize.height - layoutMeasurement.height - contentOffset.y < 100 && cursor && !loadingMore)
      void load(false, true);
  };
  return (
    <Page onRefresh={() => void load(true, false)} refreshing={refreshing} onScroll={onScroll}>
      <TopBar
        brandNameOnly
        right={
          <View style={styles.row}>
            <Button title="Search" small variant="quiet" icon="search-outline" onPress={() => router.push("/search")} />
            <Pressable
              accessibilityRole="button"
              accessibilityLabel="Open your profile"
              onPress={() => router.push("/(tabs)/profile")}
            >
              <Avatar name={app.user.name} image={app.user.avatar} size={34} />
            </Pressable>
          </View>
        }
      />
      <View style={[styles.rowBetween, { paddingVertical: 2 }]}>
        <SectionTitle title="Your feed" />
        <Pressable accessibilityRole="button" onPress={() => setQuickMenu((value) => !value)}>
          <Ionicons name="ellipsis-horizontal" color={colors.muted} size={22} />
        </Pressable>
      </View>
      {quickMenu ? (
        <View style={[styles.row, { flexWrap: "wrap" }]}>
          <Button title="Notifications" small variant="outline" onPress={() => router.push("/notifications")} />
          <Button title="Messages" small variant="outline" onPress={() => router.push("/messages")} />
          <Button title="Profile" small variant="outline" onPress={() => router.push("/(tabs)/profile")} />
          <Text style={styles.caption}>{hasUnread ? "You have unread notifications" : ""}</Text>
        </View>
      ) : null}
      <View style={styles.row}>
        <Chip label="For You" selected={mode === "for-you"} onPress={() => setMode("for-you")} />
        <Chip label="Following" selected={mode === "following"} onPress={() => setMode("following")} />
      </View>
      {error ? (
        <View accessibilityRole="alert" style={{ gap: 8, padding: 12, backgroundColor: colors.orangeTint, borderRadius: 14 }}>
          <Text style={styles.body}>Feed error: {error}</Text>
          <Button title="Retry" small variant="outline" onPress={() => void load(true)} />
        </View>
      ) : null}
      {loading ? <LoadingState label="Loading feed from the server" /> : null}
      {!loading && !error && posts.length === 0 ? (
        <EmptyState
          title={mode === "following" ? "No posts from people you follow" : "Your feed is empty"}
          detail={
            mode === "following"
              ? "Follow a person or business to see their posts here."
              : "New posts will appear here as they are published."
          }
          icon="albums-outline"
          action="Create a post"
          onAction={() => router.push("/(tabs)/create")}
        />
      ) : null}
      {posts.map((post) => (
        <ApiPostCard key={post.id} post={post} />
      ))}
      {loadingMore ? <LoadingState label="Loading more posts" /> : null}
      {!loading && posts.length > 0 && !cursor && !loadingMore ? (
        <Text style={[styles.caption, { textAlign: "center" }]}>Youâ€™re all caught up.</Text>
      ) : null}
    </Page>
  );
}

export function ApiPostComposer() {
  const services = useMabizServices();
  const app = useAppStore();
  const [step, setStep] = useState<1 | 2 | 3>(1);
  const [caption, setCaption] = useState("");
  const [kind, setKind] = useState<"text" | "image" | "video" | "product" | "announcement">("text");
  const [identityKind, setIdentityKind] = useState<"personal" | "business">("personal");
  const [businessId, setBusinessId] = useState("");
  const [audience, setAudience] = useState<"public" | "followers" | "private">("public");
  const [allowComments, setAllowComments] = useState(true);
  const [location, setLocation] = useState("");
  const [urls, setUrls] = useState("");
  const [mediaType, setMediaType] = useState<"image" | "video">("image");
  const [altText, setAltText] = useState("");
  const [pollQuestion, setPollQuestion] = useState("");
  const [pollOptions, setPollOptions] = useState(["", ""]);
  const [showPoll, setShowPoll] = useState(false);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");

  const ownedBusinesses = app.businesses.filter(
    (b) => b.ownerId === app.user.id || b.team.some((m) => m.userId === app.user.id)
  );
  const activeBusiness = ownedBusinesses.find((b) => b.id === businessId);

  const validateUrl = (url: string) => {
    try {
      const parsed = new URL(url);
      return parsed.protocol === "https:" && !parsed.username && !parsed.password;
    } catch {
      return false;
    }
  };

  const toggleIdentity = () => {
    if (identityKind === "personal") {
      setIdentityKind("business");
      if (!businessId && ownedBusinesses[0]) setBusinessId(ownedBusinesses[0].id);
    } else {
      setIdentityKind("personal");
    }
  };

  const publish = async () => {
    const cleanCaption = caption.trim();
    if (!cleanCaption) {
      setError("Write a caption before publishing.");
      return;
    }
    const mediaUrls = urls
      .split(/\n|,/)
      .map((url) => url.trim())
      .filter(Boolean);
    if (mediaUrls.some((url) => !validateUrl(url))) {
      setError("Every media reference must be a hosted HTTPS URL.");
      return;
    }
    if (identityKind === "business" && !businessId.trim()) {
      setError("Select or enter a Business ID to publish as a business.");
      return;
    }
    const options = pollOptions.map((item) => item.trim()).filter(Boolean);
    const wantsPoll = showPoll && pollQuestion.trim().length > 0;
    if (wantsPoll && (options.length < 2 || options.length > 6)) {
      setError("A poll requires 2â€“6 non-empty options.");
      return;
    }
    const identityId = identityKind === "personal" ? app.user.id : businessId.trim();
    setBusy(true);
    setError("");
    const result = await services.posts.create({
      kind,
      caption: cleanCaption,
      identityKind,
      identityId,
      businessId: identityKind === "business" ? identityId : null,
      audience,
      allowComments,
      location: location.trim() || undefined,
      hashtags: [...new Set((cleanCaption.match(/#([\p{L}\p{N}_]+)/gu) ?? []).map((tag) => tag.slice(1)))],
      mediaReferences: mediaUrls.map((url) => ({
        mediaType,
        url,
        ...(altText.trim() ? { altText: altText.trim() } : {}),
      })),
    });
    if (!result.ok) {
      setError(errorText(result.error));
      setBusy(false);
      return;
    }
    if (wantsPoll) {
      const closesAt = new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString();
      const pollInput: ApiPollInput = { question: pollQuestion.trim(), options, closesAt };
      const pollResult = await services.posts.createPoll(result.data.id, pollInput);
      if (!pollResult.ok) {
        setError(`Post published, but poll creation failed: ${errorText(pollResult.error)}`);
        setBusy(false);
        router.replace({ pathname: "/post/[id]", params: { id: result.data.id } });
        return;
      }
    }
    setBusy(false);
    router.replace("/(tabs)");
  };

  return (
    <Page>
      {/* Top Navigation Bar */}
      <View style={{ flexDirection: "row", alignItems: "center", justifyContent: "space-between", paddingVertical: 8 }}>
        <Pressable
          onPress={() => (step > 1 ? setStep((s) => (s === 3 ? 2 : 1)) : router.back())}
          style={{ padding: 4 }}
        >
          <Ionicons name="arrow-back" size={22} color={colors.navy} />
        </Pressable>
        <Text style={{ color: colors.navy, fontSize: 17, fontWeight: "800" }}>Create post</Text>
        <Pressable onPress={() => router.push("/drafts")}>
          <Text style={{ color: colors.emeraldDark, fontSize: 14, fontWeight: "700" }}>Drafts</Text>
        </Pressable>
      </View>

      {!app.isAuthenticated ? (
        <EmptyState
          title="Sign in to publish"
          detail="Publishing requires an authenticated API session."
          icon="lock-closed-outline"
          action="Sign in"
          onAction={() => router.push("/auth/login")}
        />
      ) : null}

      {/* ================= STEP 1: COMPOSE ================= */}
      {step === 1 && (
        <View style={{ gap: 14, paddingTop: 4 }}>
          <Pressable onPress={toggleIdentity} style={{ flexDirection: "row", alignItems: "center", gap: 10 }}>
            <Avatar
              name={identityKind === "business" ? activeBusiness?.name ?? "Business" : app.user.name}
              image={identityKind === "business" ? activeBusiness?.logo : app.user.avatar}
              size={42}
            />
            <View style={{ flex: 1 }}>
              <Text style={{ color: colors.navy, fontSize: 15, fontWeight: "800" }}>
                {identityKind === "business" ? activeBusiness?.name ?? "Business profile" : app.user.name}
              </Text>
              <View
                style={{
                  alignSelf: "flex-start",
                  flexDirection: "row",
                  alignItems: "center",
                  gap: 4,
                  marginTop: 2,
                  paddingHorizontal: 10,
                  paddingVertical: 3,
                  borderRadius: 12,
                  backgroundColor: colors.subtle,
                }}
              >
                <Text style={{ color: colors.muted, fontSize: 12, fontWeight: "600" }}>
                  {identityKind === "business" ? "Posting as business" : "Posting to your profile"}
                </Text>
                <Ionicons name="chevron-down" size={12} color={colors.muted} />
              </View>
            </View>
          </Pressable>

          {identityKind === "business" ? (
            <View style={{ gap: 6 }}>
              {ownedBusinesses.length > 0 ? (
                <View style={{ flexDirection: "row", flexWrap: "wrap", gap: 6 }}>
                  {ownedBusinesses.map((b) => (
                    <Chip key={b.id} label={b.name} selected={businessId === b.id} onPress={() => setBusinessId(b.id)} />
                  ))}
                </View>
              ) : null}
              <Field
                label="Business ID"
                value={businessId}
                onChangeText={setBusinessId}
                placeholder="Enter business ID"
                autoCapitalize="none"
              />
            </View>
          ) : null}

          <TextInput
            style={{
              color: colors.navy,
              fontSize: 16,
              minHeight: 170,
              textAlignVertical: "top",
              paddingTop: 4,
            }}
            placeholder="What would you like to share?"
            placeholderTextColor={colors.muted}
            multiline
            maxLength={1000}
            value={caption}
            onChangeText={setCaption}
          />

          {showPoll ? (
            <View style={{ gap: 8, padding: 12, borderRadius: 14, backgroundColor: colors.subtle }}>
              <Field
                label="Poll question"
                value={pollQuestion}
                onChangeText={setPollQuestion}
                placeholder="Ask your audience"
              />
              {pollOptions.map((item, index) => (
                <Field
                  key={index}
                  placeholder={`Option ${index + 1}`}
                  value={item}
                  onChangeText={(value) =>

                />

                title="Add poll option"

                variant="outline"
                disabled={pollOptions.length >= 6}

          ) : null}

          <View style={{ alignItems: "flex-end" }}>
            <Text style={{ color: colors.muted, fontSize: 11, fontVariant: ["tabular-nums"] }}>
              {caption.length}/1000
            </Text>
          </View>

          <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ gap: 8, paddingVertical: 4 }}>
            {(
              [

                { label: "Poll", value: "text", icon: "stats-chart-outline", poll: true },

          <View

              paddingVertical: 12,

                  setMediaType("video");

                  setStep(2);

                key={tool.label}
                onPress={tool.onPress}

                  alignItems: "center",
                  gap: 4,
                  flex: 1,
                  paddingVertical: 8,
                  borderRadius: 12,
                  backgroundColor: colors.subtle,
                  marginHorizontal: 4,

              >

                <Text style={{ color: colors.navy, fontSize: 11, fontWeight: "600" }}>{tool.label}</Text>

            }}

                </Text>
                <Text style={{ color: colors.muted, fontSize: 11, textTransform: "capitalize" }}>
                  {audience} {location ? `Â· ${location}` : ""}
                </Text>
              </View>
            </View>

            <Text style={{ color: colors.navy, fontSize: 14, lineHeight: 20 }}>
              {caption.trim() || "No caption written yet."}
            </Text>

            {urls.trim() ? (
              <Text style={{ color: colors.emeraldDark, fontSize: 12 }}>

              </Text>
            ) : null}
          </View>

          <View style={{ borderWidth: 1, borderColor: colors.border, borderRadius: 16, overflow: "hidden" }}>
            <View
              style={{
                flexDirection: "row",
                justifyContent: "space-between",
                padding: 14,
                backgroundColor: colors.surface,
                borderBottomWidth: 1,
                borderBottomColor: colors.border,
              }}
            >
              <Text style={{ color: colors.navy, fontSize: 14, fontWeight: "600" }}>Audience</Text>
              <Text style={{ color: colors.muted, fontSize: 14, fontWeight: "600", textTransform: "capitalize" }}>
                {audience}
              </Text>
            </View>
            <View
              style={{
                flexDirection: "row",
                justifyContent: "space-between",
                padding: 14,
                backgroundColor: colors.surface,
              }}
            >
              <Text style={{ color: colors.navy, fontSize: 14, fontWeight: "600" }}>Comments</Text>
              <Text style={{ color: colors.muted, fontSize: 14, fontWeight: "600" }}>
                {allowComments ? "On" : "Off"}
              </Text>
            </View>
          </View>

          {error ? (
            <Text accessibilityRole="alert" style={{ ...styles.body, color: colors.danger }}>
              Publish error: {error}
            </Text>
          ) : null}

          <View style={{ gap: 8, marginTop: 8 }}>
            <Button
              title={busy ? "Sharingâ€¦" : "Share post"}
              disabled={busy || !app.isAuthenticated}
              onPress={() => void publish()}
            />
            <Button title="Back" variant="outline" onPress={() => setStep(2)} />
          </View>
        </View>
      )}
    </Page>
  );
}

export function ApiPostDetail({ postId }: { postId: string }) {
  const services = useMabizServices();
  const app = useAppStore();
  const [post, setPost] = useState<Post | null>(null);
  const [comments, setComments] = useState<SocialApiComment[]>([]);
  const [commentText, setCommentText] = useState("");
  const [replyTo, setReplyTo] = useState<SocialApiComment | null>(null);
  const [loading, setLoading] = useState(true);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const [commentError, setCommentError] = useState("");
  const load = useCallback(async () => {
    setLoading(true);
    setError("");
    setCommentError("");
    const [postResult, commentsResult] = await Promise.all([
      services.posts.get(postId),
      services.posts.listComments(postId, pageRequest()),
    ]);
    if (!postResult.ok) setError(errorText(postResult.error));
    else setPost(postResult.data);
    if (!commentsResult.ok) setCommentError(errorText(commentsResult.error));
    else setComments(commentsResult.data.items);
    setLoading(false);
  }, [postId, services]);
  useEffect(() => {
    void load();
  }, [load]);
  const send = async () => {
    if (!app.isAuthenticated) {
      router.push("/auth/login");
      return;
    }

import type { CatalogMediaUploadInput } from "@/lib/api/feature-contracts";

/** React Native hands multipart parts to the native HTTP stack as a plain object, not as a Blob. */
export interface NativeFilePart {
  uri: string;
  name: string;
  type: string;
}

export type CatalogMediaRuntime = "web" | "native";

export interface CatalogUploadIssue {
  code: string;
  message: string;
}

/** The Stage 5 server accepts exactly these three image types. */
export const SUPPORTED_CATALOGUE_IMAGE_TYPES: readonly string[] = ["image/jpeg", "image/jpg", "image/png", "image/webp"];
export const MAX_CATALOGUE_GALLERY_IMAGES = 12;
/**
 * Hard ceiling of the server's configurable 1-64 MiB upload limit. The authoritative cap is the
 * deployment's Storage:MaximumUploadMebibytes (default 10), which the server enforces and reports as
 * MEDIA_TOO_LARGE; pre-checking only at the ceiling avoids blocking a file a deployment would accept.
 */
export const MAX_CATALOGUE_IMAGE_BYTES = 64 * 1024 * 1024;

const DEVICE_URI_PATTERN = /^(file|content|blob|asset|ph|data):/i;

/** True for a path that only exists on this device (or in the preview sandbox). */
export function isDeviceFileUri(value?: string | null): boolean {
  if (!value) return false;
  return DEVICE_URI_PATTERN.test(value) || value.startsWith("/manus-storage/");
}

export function isSupportedCatalogueImage(contentType?: string | null): boolean {
  if (!contentType) return false;
  return SUPPORTED_CATALOGUE_IMAGE_TYPES.includes(contentType.trim().toLowerCase());
}

/** Client-side mirror of the server intake rules, so a doomed upload is explained with the server's own code. */
export function catalogUploadIssue(file: CatalogMediaUploadInput): CatalogUploadIssue | null {
  if (!file.uri?.trim()) return { code: "MEDIA_EMPTY", message: "Choose an image from this device before uploading." };
  if (!isSupportedCatalogueImage(file.contentType)) return { code: "MEDIA_TYPE_NOT_SUPPORTED", message: "Catalogue images must be JPEG, PNG or WebP." };
  if (file.sizeBytes !== undefined && file.sizeBytes > MAX_CATALOGUE_IMAGE_BYTES) return { code: "MEDIA_TOO_LARGE", message: "Catalogue images cannot exceed 64 MB. The server applies its own configured limit and will reject anything larger." };
  return null;
}

export function detectCatalogMediaRuntime(): CatalogMediaRuntime {
  return typeof document !== "undefined" && typeof window !== "undefined" ? "web" : "native";
}

function safeFileName(file: CatalogMediaUploadInput): string {
  const name = file.fileName?.trim();
  if (name) return name;
  const type = file.contentType?.toLowerCase();
  const extension = type === "image/png" ? "png" : type === "image/webp" ? "webp" : "jpg";
  return `catalogue-image.${extension}`;
}

/** The part shape the React Native FormData implementation expects. */
export function catalogNativeFilePart(file: CatalogMediaUploadInput): NativeFilePart {
  return { uri: file.uri, name: safeFileName(file), type: file.contentType || "image/jpeg" };
}

/**
 * Builds the `multipart/form-data` body for `POST /businesses/{businessId}/media`.
 * Web reads the picked object URL into a Blob; native appends the device part directly.
 */
export async function buildCatalogMediaForm(
  file: CatalogMediaUploadInput,
  runtime: CatalogMediaRuntime = detectCatalogMediaRuntime(),
  fetcher: typeof fetch = fetch,
): Promise<FormData> {
  const form = new FormData();
  if (runtime === "web") {
    const response = await fetcher(file.uri);
    if (!response.ok) throw new Error("The selected image could not be read from this device.");
    form.append("file", await response.blob(), safeFileName(file));
    return form;
  }
  form.append("file", catalogNativeFilePart(file) as unknown as Blob);
  return form;
}

import { apiConfig, type ApiClientConfig } from "@/lib/api/config";
import type { ApiError, HttpMethod, QueryParams, QueryValue, Result } from "@/lib/api/contracts";

export interface AuthTokenProvider {
  getAccessToken(): Promise<string | null>;
  refreshAccessToken(): Promise<string | null>;
  clearTokens?(): Promise<void>;
  onSessionExpired?(): Promise<void> | void;
}

export interface HttpRequestOptions extends Omit<RequestInit, "body" | "headers" | "method"> {
  headers?: HeadersInit;
  body?: unknown;
  query?: QueryParams;
  authenticated?: boolean;
  timeoutMs?: number;
  requestId?: string;
}

function createRequestId(): string {
  const random = Math.random().toString(36).slice(2, 10);
  return `bc-${Date.now().toString(36)}-${random}`;
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function appendQuery(params: URLSearchParams, key: string, value: QueryValue): void {
  if (value === undefined || value === null) return;
  if (Array.isArray(value)) {
    for (const item of value) if (item !== undefined && item !== null) params.append(key, String(item));
    return;
  }
  params.append(key, String(value));
}

function buildUrl(config: ApiClientConfig, path: string, query?: QueryParams): string {
  if (!path.startsWith("/") || path.startsWith("//") || path.includes("?") || path.includes("#")) {
    throw new Error("API paths must be relative paths without a query string or fragment.");
  }
  const origin = new URL(config.baseUrl);
  if (origin.protocol !== "https:" && !(origin.protocol === "http:" && config.allowInsecureHttp)) {
    throw new Error("The API origin must use HTTPS outside an explicitly allowed development connection.");
  }
  const prefix = config.versionPrefix ? `/${config.versionPrefix.replace(/^\/+|\/+$/g, "")}` : "";
  const url = new URL(`${origin.origin}${prefix}${path}`);
  for (const [key, value] of Object.entries(query ?? {})) appendQuery(url.searchParams, key, value);
  return url.toString();
}

function isFormData(value: unknown): value is FormData {
  return typeof FormData !== "undefined" && value instanceof FormData;
}

function isRawBody(value: unknown): value is BodyInit {
  if (typeof value === "string") return true;
  if (isFormData(value)) return true;
  if (typeof URLSearchParams !== "undefined" && value instanceof URLSearchParams) return true;
  if (typeof Blob !== "undefined" && value instanceof Blob) return true;
  if (typeof ArrayBuffer !== "undefined" && (value instanceof ArrayBuffer || ArrayBuffer.isView(value))) return true;
  return false;
}

function prepareBody(body: unknown, headers: Headers): BodyInit | undefined {
  if (body === undefined || body === null) return undefined;
  if (isFormData(body)) return body;
  if (isRawBody(body)) {
    if (typeof body === "string" && !headers.has("Content-Type")) headers.set("Content-Type", "text/plain;charset=UTF-8");
    return body;
  }
  if (!headers.has("Content-Type")) headers.set("Content-Type", "application/json");
  return JSON.stringify(body);
}

function retryableStatus(status: number): boolean {
  return status === 408 || status === 425 || status === 429 || status >= 500;
}

function responseError(status: number, payload: unknown, requestId: string): ApiError {
  const root = isRecord(payload) ? payload : {};
  const envelopeError = isRecord(root.error) ? root.error : {};
  const detailsValue = envelopeError.details ?? root.details ?? root.errors;
  const details = isRecord(detailsValue) ? detailsValue : undefined;
  const defaultCode = status === 401 ? "UNAUTHENTICATED"
    : status === 403 ? "FORBIDDEN"
      : status === 404 ? "NOT_FOUND"
        : status === 409 ? "CONFLICT"
          : status === 422 ? "VALIDATION_ERROR"
            : status === 429 ? "RATE_LIMITED"
              : status >= 500 ? "SERVER_ERROR" : `HTTP_${status}`;
  const code = typeof envelopeError.code === "string" ? envelopeError.code
    : typeof root.code === "string" ? root.code : defaultCode;
  const message = typeof envelopeError.message === "string" ? envelopeError.message
    : typeof root.message === "string" ? root.message : "The request could not be completed.";
  return {
    code,
    message,
    retryable: typeof envelopeError.retryable === "boolean" ? envelopeError.retryable : retryableStatus(status),
    status,
    requestId,
    ...(details ? { details } : {}),
  };
}

async function readPayload(response: Response): Promise<{ value: unknown; invalidJson: boolean }> {
  if (response.status === 204 || response.status === 205) return { value: undefined, invalidJson: false };
  const text = await response.text();
  if (!text) return { value: undefined, invalidJson: false };
  const contentType = response.headers.get("content-type")?.toLowerCase() ?? "";
  if (contentType.includes("json") || /^[\[{]/.test(text.trim())) {
    try {
      return { value: JSON.parse(text) as unknown, invalidJson: false };
    } catch {
      return { value: undefined, invalidJson: true };
    }
  }
  return { value: text, invalidJson: false };
}

function responseRequestId(response: Response, payload: unknown, fallback: string): string {
  const headerId = response.headers.get("x-request-id") ?? response.headers.get("request-id");
  if (headerId) return headerId;
  return isRecord(payload) && typeof payload.requestId === "string" ? payload.requestId : fallback;
}

function errorResult<T>(requestId: string, code: string, message: string, retryable = false, details?: Record<string, unknown>, status?: number): Result<T> {
  return { ok: false, source: "api", requestId, error: { code, message, retryable, requestId, ...(status !== undefined ? { status } : {}), ...(details ? { details } : {}) } };
}

/** Central JSON transport. The mock-first application does not call it until API mode is explicitly configured. */
export class JsonApiClient {
  private refreshInFlight: Promise<string | null> | null = null;

  constructor(
    private readonly tokens?: AuthTokenProvider,
    private readonly request: typeof fetch = fetch,
    private readonly timeoutMs = apiConfig.timeoutMs,
    private readonly config: ApiClientConfig = apiConfig,
  ) {}

  get<T>(path: string, options: HttpRequestOptions = {}): Promise<Result<T>> {
    return this.send<T>("GET", path, options);
  }

  post<T>(path: string, body?: unknown, options: HttpRequestOptions = {}): Promise<Result<T>> {
    return this.send<T>("POST", path, { ...options, ...(body !== undefined ? { body } : {}) });
  }

  put<T>(path: string, body?: unknown, options: HttpRequestOptions = {}): Promise<Result<T>> {
    return this.send<T>("PUT", path, { ...options, ...(body !== undefined ? { body } : {}) });
  }

  patch<T>(path: string, body?: unknown, options: HttpRequestOptions = {}): Promise<Result<T>> {
    return this.send<T>("PATCH", path, { ...options, ...(body !== undefined ? { body } : {}) });
  }

  delete<T>(path: string, options: HttpRequestOptions = {}): Promise<Result<T>> {
    return this.send<T>("DELETE", path, options);
  }

  async send<T>(method: HttpMethod, path: string, options: HttpRequestOptions = {}): Promise<Result<T>> {
    const requestId = options.requestId ?? createRequestId();
    if (this.config.mode !== "api") return errorResult(requestId, "API_MODE_DISABLED", "The API service is disabled; this app is using its local demo services.");
    if (!this.config.baseUrl.trim()) return errorResult(requestId, "API_NOT_CONFIGURED", "Set an API origin before enabling production service calls.");

    let url: string;
    try {
      url = buildUrl(this.config, path, options.query);
    } catch (error) {
      return errorResult(requestId, "INVALID_API_CONFIGURATION", error instanceof Error ? error.message : "The API request configuration is invalid.");
    }

    const authenticated = options.authenticated !== false;
    let token: string | null = null;
    if (authenticated && this.tokens) {
      try {
        token = await this.tokens.getAccessToken();
      } catch {
        return errorResult(requestId, "AUTH_TOKEN_UNAVAILABLE", "The secure session could not be read.");
      }
    }

    const first = await this.perform<T>(method, url, options, requestId, token);
    if (first.ok || first.error.status !== 401 || !authenticated || !this.tokens) return first;

    let refreshed: string | null = null;
    try {
      refreshed = await this.refreshTokenSingleFlight();
    } catch {
      refreshed = null;
    }
    if (!refreshed) {
      await this.expireSession();
      return errorResult(requestId, "SESSION_EXPIRED", "Your session has expired. Sign in again to continue.");
    }

    const retried = await this.perform<T>(method, url, options, requestId, refreshed);
    if (!retried.ok && retried.error.status === 401) {
      await this.expireSession();
      return errorResult(requestId, "SESSION_EXPIRED", "Your session has expired. Sign in again to continue.");
    }
    return retried;
  }

  private refreshTokenSingleFlight(): Promise<string | null> {
    if (!this.tokens) return Promise.resolve(null);
    if (!this.refreshInFlight) {
      const refresh = this.tokens.refreshAccessToken();
      this.refreshInFlight = refresh;
      void refresh.finally(() => {
        if (this.refreshInFlight === refresh) this.refreshInFlight = null;
      }).catch(() => undefined);
    }
    return this.refreshInFlight;
  }

  private async expireSession(): Promise<void> {
    await this.tokens?.clearTokens?.().catch(() => undefined);
    try {
      await this.tokens?.onSessionExpired?.();
    } catch {
      // Session cleanup is best-effort; the original API failure remains authoritative.
    }
  }

  private async perform<T>(method: HttpMethod, url: string, options: HttpRequestOptions, requestId: string, token: string | null): Promise<Result<T>> {
    const controller = new AbortController();
    const timeoutMs = options.timeoutMs ?? this.timeoutMs;
    let timedOut = false;
    const callerSignal = options.signal;
    const abortFromCaller = () => controller.abort();
    if (callerSignal?.aborted) controller.abort();
    else callerSignal?.addEventListener("abort", abortFromCaller, { once: true });
    const timeout = setTimeout(() => { timedOut = true; controller.abort(); }, timeoutMs);

    try {
      if (callerSignal?.aborted) return errorResult(requestId, "REQUEST_CANCELLED", "The request was cancelled.");
      const headers = new Headers(options.headers);
      headers.set("Accept", "application/json");
      headers.set("X-Request-ID", requestId);
      if (options.authenticated !== false && token) headers.set("Authorization", `Bearer ${token}`);
      const body = prepareBody(options.body, headers);
      const { headers: _headers, body: _body, query: _query, authenticated: _authenticated, timeoutMs: _timeoutMs, requestId: _requestId, signal: _signal, ...init } = options;
      const response = await this.request(url, { ...init, method, headers, ...(body !== undefined ? { body } : {}), signal: controller.signal });
      const parsed = await readPayload(response);
      const payload = parsed.value;
      const serverRequestId = responseRequestId(response, payload, requestId);

      if (parsed.invalidJson && response.ok) {
        return errorResult(serverRequestId, "INVALID_JSON_RESPONSE", "The API returned an unreadable JSON response.", false, undefined, response.status);
      }

      if (!response.ok) {
        const root = isRecord(payload) ? payload : {};
        const envelopeError = isRecord(root.error) ? root.error : undefined;
        if (root.ok === false || envelopeError || response.status >= 400) {
          return { ok: false, source: "api", requestId: serverRequestId, error: responseError(response.status, payload, serverRequestId) };
        }
      }

      if (isRecord(payload) && payload.ok === false) {
        const status = typeof payload.status === "number" ? payload.status : 400;
        return { ok: false, source: "api", requestId: serverRequestId, error: responseError(status, payload, serverRequestId) };
      }
      const data = isRecord(payload) && payload.ok === true && "data" in payload ? payload.data : payload;
      return { ok: true, source: "api", requestId: serverRequestId, data: data as T };
    } catch (error) {
      if (timedOut) return errorResult(requestId, "REQUEST_TIMEOUT", "The request timed out.", true, { timeoutMs });
      if (callerSignal?.aborted) return errorResult(requestId, "REQUEST_CANCELLED", "The request was cancelled.");
      if (error instanceof TypeError && /url|fetch|network/i.test(error.message)) {
        return errorResult(requestId, "NETWORK_ERROR", "The API could not be reached.", true);
      }
      return errorResult(requestId, "NETWORK_ERROR", "The API request failed before a response was received.", true);
    } finally {
      clearTimeout(timeout);
      callerSignal?.removeEventListener("abort", abortFromCaller);
    }
  }
}

import { HubConnection, HubConnectionBuilder, HubConnectionState, LogLevel } from "@microsoft/signalr";
import { apiConfig } from "@/lib/api/config";
import { SecureSessionTokenProvider } from "@/lib/api/secure-session";
import type { AuthSession } from "@/lib/api/feature-contracts";
import type { ChatMessage } from "@/types/domain";

export type MessagingConnectionState = "connecting" | "connected" | "reconnecting" | "disconnected";
export interface RealtimeNotificationEvent { id: string; type: string; createdAt: string; }

/** SignalR carries events only. REST remains authoritative and callers resync history after reconnect. */
export class MessagingRealtimeClient {
  private connection: HubConnection;
  private joinedConversation: string | null = null;
  private messageListeners = new Set<(message: ChatMessage) => void>();
  private stateListeners = new Set<(state: MessagingConnectionState) => void>();
  private readListeners = new Set<(readerUserId: string, readAt: string) => void>();
  private statusListeners = new Set<(messageId: string, status: "delivered" | "read") => void>();
  private notificationListeners = new Set<(notification: RealtimeNotificationEvent) => void>();
  private disposed = false;

  constructor(baseUrl = apiConfig.baseUrl) {
    if (!baseUrl) throw new Error("API_NOT_CONFIGURED");
    const tokens = new SecureSessionTokenProvider();
    const refreshPath = apiConfig.authRefreshPath ?? "/auth/refresh";
    tokens.setRefreshHandler(async (refreshToken): Promise<AuthSession | null> => {
      try {
        const response = await fetch(`${baseUrl.replace(/\/$/, "")}${apiConfig.versionPrefix}${refreshPath}`, {
          method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ refreshToken }),
        });
        return response.ok ? await response.json() as AuthSession : null;
      } catch { return null; }
    });
    this.connection = new HubConnectionBuilder()
      .withUrl(`${baseUrl.replace(/\/$/, "")}/hubs/messaging`, {
        accessTokenFactory: async () => {
          const session = await tokens.getSession();
          if (!session) return "";
          if (Date.parse(session.expiresAt) <= Date.now() + 60_000)
            return await tokens.refreshAccessToken() ?? "";
          return session.accessToken;
        },
      })
      .withAutomaticReconnect([0, 2_000, 5_000, 10_000])
      .configureLogging(__DEV__ ? LogLevel.Warning : LogLevel.Error)
      .build();
    this.connection.on("MessageReceived", (value: ApiHubMessage) => {
      const message: ChatMessage = { id: value.id, senderId: value.senderUserId, text: value.text, createdAt: value.createdAt, status: value.status };
      for (const listener of this.messageListeners) listener(message);
    });
    this.connection.on("ConversationRead", (value: { readerUserId: string; readAt: string }) => {
      for (const listener of this.readListeners) listener(value.readerUserId, value.readAt);
    });
    this.connection.on("MessageStatusChanged", (value: { messageId: string; status: "delivered" | "read" }) => {
      for (const listener of this.statusListeners) listener(value.messageId, value.status);
    });
    this.connection.on("NotificationReceived", (value: RealtimeNotificationEvent) => {
      for (const listener of this.notificationListeners) listener(value);
    });
    this.connection.onreconnecting(() => this.emitState("reconnecting"));
    this.connection.onreconnected(async () => {
      this.emitState("connected");
      if (this.joinedConversation) await this.connection.invoke("JoinConversation", this.joinedConversation).catch(() => undefined);
    });
    this.connection.onclose(() => this.emitState("disconnected"));
  }

  async connect(): Promise<void> {
    if (this.disposed || this.connection.state !== HubConnectionState.Disconnected) return;
    this.emitState("connecting");
    await this.connection.start();
    this.emitState("connected");
    if (this.joinedConversation) await this.connection.invoke("JoinConversation", this.joinedConversation);
  }

  async joinConversation(conversationId: string): Promise<void> {
    this.joinedConversation = conversationId;
    if (this.connection.state === HubConnectionState.Connected) await this.connection.invoke("JoinConversation", conversationId);
  }

  async leaveConversation(conversationId: string): Promise<void> {
    if (this.joinedConversation === conversationId) this.joinedConversation = null;
    if (this.connection.state === HubConnectionState.Connected) await this.connection.invoke("LeaveConversation", conversationId);
  }

  async sendMessage(conversationId: string, clientMessageId: string, text: string): Promise<ChatMessage> {
    if (this.connection.state !== HubConnectionState.Connected) throw new Error("SIGNALR_DISCONNECTED");
    const response = await this.connection.invoke<ApiHubMessage>("SendMessage", conversationId, clientMessageId, text);
    return { id: response.id, senderId: response.senderUserId, text: response.text, createdAt: response.createdAt, status: response.status };
  }

  onMessage(listener: (message: ChatMessage) => void): () => void { this.messageListeners.add(listener); return () => this.messageListeners.delete(listener); }
  onRead(listener: (readerUserId: string, readAt: string) => void): () => void { this.readListeners.add(listener); return () => this.readListeners.delete(listener); }
  onStatus(listener: (messageId: string, status: "delivered" | "read") => void): () => void { this.statusListeners.add(listener); return () => this.statusListeners.delete(listener); }
  onNotification(listener: (notification: RealtimeNotificationEvent) => void): () => void { this.notificationListeners.add(listener); return () => this.notificationListeners.delete(listener); }
  onState(listener: (state: MessagingConnectionState) => void): () => void { this.stateListeners.add(listener); return () => this.stateListeners.delete(listener); }

  async stop(): Promise<void> {
    this.disposed = true;
    this.joinedConversation = null;
    await this.connection.stop();
    this.emitState("disconnected");
  }

  private emitState(state: MessagingConnectionState): void { for (const listener of this.stateListeners) listener(state); }
}

export interface UploadTransportOptions {
  signal?: AbortSignal;
  onProgress?: (progress: MediaUploadProgress) => void;
}

class UploadTransport {
  private readonly active = new Map<string, () => Promise<void>>();

  constructor(private readonly allowInsecureHttp: boolean) {}

  private isSecureUrl(raw: string): boolean {
    try {
      const parsed = new URL(raw);
      return !parsed.username && !parsed.password && (parsed.protocol === "https:" || (parsed.protocol === "http:" && this.allowInsecureHttp));
    } catch {
      return false;
    }
  }

  async send(ticket: MediaUploadTicket, file: MediaFileInput, options: UploadTransportOptions = {}): Promise<ApiError | null> {
    if (!file.uri || !file.fileName || file.sizeBytes < 0) return { code: "INVALID_MEDIA_FILE", message: "The selected file is not available for upload.", retryable: false };
    if (!this.isSecureUrl(ticket.uploadUrl)) return { code: "INVALID_UPLOAD_URL", message: "The upload ticket did not contain a secure storage URL.", retryable: false };
    if (options.signal?.aborted) return { code: "REQUEST_CANCELLED", message: "The upload was cancelled.", retryable: false };
    if (typeof document !== "undefined" && typeof window !== "undefined") {
      return this.sendBrowser(ticket, file, options);
    }
    return this.sendNative(ticket, file, options);
  }

  private async sendBrowser(ticket: MediaUploadTicket, file: MediaFileInput, options: UploadTransportOptions): Promise<ApiError | null> {
    let blob: Blob;
    try {
      const response = await fetch(file.uri);
      if (!response.ok) return { code: "LOCAL_FILE_UNAVAILABLE", message: "The selected file could not be read from this device.", retryable: false };
      blob = await response.blob();
    } catch {
      return { code: "LOCAL_FILE_UNAVAILABLE", message: "The selected file could not be read from this device.", retryable: false };
    }

    return new Promise<ApiError | null>((resolve) => {
      const xhr = new XMLHttpRequest();
      this.active.set(ticket.uploadId, async () => xhr.abort());
      const cleanup = () => {
        this.active.delete(ticket.uploadId);
        options.signal?.removeEventListener("abort", abort);
      };
      const abort = () => xhr.abort();
      options.signal?.addEventListener("abort", abort, { once: true });
      xhr.open(ticket.method, ticket.uploadUrl, true);
      for (const [key, value] of Object.entries(ticket.headers ?? {})) xhr.setRequestHeader(key, value);
      xhr.upload.onprogress = (event) => {
        const totalBytes = event.lengthComputable ? event.total : file.sizeBytes;
        const bytesSent = event.loaded;
        options.onProgress?.({ bytesSent, totalBytes, percent: totalBytes > 0 ? Math.min(100, Math.round((bytesSent / totalBytes) * 100)) : 0 });
      };
      xhr.onload = () => {
        cleanup();
        if (xhr.status >= 200 && xhr.status < 300) {
          options.onProgress?.({ bytesSent: file.sizeBytes, totalBytes: file.sizeBytes, percent: 100 });
          resolve(null);
        } else {
          resolve(this.uploadError(xhr.status, xhr.responseText));
        }
      };
      xhr.onerror = () => { cleanup(); resolve({ code: "UPLOAD_NETWORK_ERROR", message: "The media upload could not reach storage.", retryable: true }); };
      xhr.onabort = () => { cleanup(); resolve({ code: "UPLOAD_CANCELLED", message: "The media upload was cancelled.", retryable: true }); };

      if (ticket.transfer === "binary") {
        xhr.setRequestHeader("Content-Type", file.contentType);
        xhr.send(blob);
        return;
      }
      const form = new FormData();
      for (const [key, value] of Object.entries(ticket.fields ?? {})) form.append(key, value);
      form.append(ticket.fieldName || "file", blob, file.fileName);
      xhr.send(form);
    });
  }

  private async sendNative(ticket: MediaUploadTicket, file: MediaFileInput, options: UploadTransportOptions): Promise<ApiError | null> {
    try {
      const FileSystem = await import("expo-file-system/legacy");
      const uploadType = ticket.transfer === "binary" ? FileSystem.FileSystemUploadType.BINARY_CONTENT : FileSystem.FileSystemUploadType.MULTIPART;
      const task = FileSystem.createUploadTask(ticket.uploadUrl, file.uri, {
        uploadType,
        httpMethod: ticket.method,
        headers: { ...(ticket.headers ?? {}), ...(ticket.transfer === "binary" ? { "Content-Type": file.contentType } : {}) },
        sessionType: FileSystem.FileSystemSessionType.FOREGROUND,
        ...(ticket.transfer === "multipart" ? { fieldName: ticket.fieldName || "file", mimeType: file.contentType, parameters: ticket.fields ?? {} } : {}),
      }, (event) => {
        const totalBytes = event.totalBytesExpectedToSend || file.sizeBytes;
        const bytesSent = event.totalBytesSent;
        options.onProgress?.({ bytesSent, totalBytes, percent: totalBytes > 0 ? Math.min(100, Math.round((bytesSent / totalBytes) * 100)) : 0 });
      });
      this.active.set(ticket.uploadId, () => task.cancelAsync());
      const abort = () => { void task.cancelAsync(); };
      options.signal?.addEventListener("abort", abort, { once: true });
      const result = await task.uploadAsync();
      options.signal?.removeEventListener("abort", abort);
      this.active.delete(ticket.uploadId);
      if (!result) return { code: "UPLOAD_CANCELLED", message: "The media upload was cancelled.", retryable: true };
      if (result.status < 200 || result.status >= 300) return this.uploadError(result.status, result.body);
      options.onProgress?.({ bytesSent: file.sizeBytes, totalBytes: file.sizeBytes, percent: 100 });
      return null;
    } catch (error) {
      this.active.delete(ticket.uploadId);
      if (options.signal?.aborted) return { code: "UPLOAD_CANCELLED", message: "The media upload was cancelled.", retryable: true };
      const message = error instanceof Error ? error.message : "The upload failed before completion.";
      return { code: "UPLOAD_TRANSPORT_ERROR", message, retryable: true };
    }
  }

  private uploadError(status: number, body: string): ApiError {
    let code = status === 401 || status === 403 ? "UPLOAD_NOT_AUTHORIZED" : status === 413 ? "UPLOAD_TOO_LARGE" : status >= 500 ? "UPLOAD_STORAGE_ERROR" : `UPLOAD_HTTP_${status}`;
    let message = "The media could not be uploaded.";
    try {
      const payload: unknown = JSON.parse(body);
      if (payload && typeof payload === "object") {
        const root = payload as Record<string, unknown>;
        const nested = root.error && typeof root.error === "object" ? root.error as Record<string, unknown> : root;
        if (typeof nested.code === "string") code = nested.code;
        if (typeof nested.message === "string") message = nested.message;
      }
    } catch {
      // Never surface storage response bodies, signed URLs or provider diagnostics to the UI.
    }
    return { code, message, retryable: status === 408 || status === 425 || status === 429 || status >= 500, status };
  }

  async cancelLocal(uploadId: string): Promise<void> {
    await this.active.get(uploadId)?.();
    this.active.delete(uploadId);
  }
}

/**
 * Production-mode HTTP adapters. These map the frontend contracts to proposed API routes;
 * configuring them does not create a backend or make an endpoint exist.
 */
export function createApiServices(options: ApiServicesOptions = {}): MabizServices {
  const config = options.config ?? apiConfig;
  const tokens = options.tokenProvider ?? new SecureSessionTokenProvider(options.sessionStorage, options.onSessionExpired);
  const client = options.client ?? new JsonApiClient(tokens, fetch, config.timeoutMs, config);
  const uploads = new UploadTransport(config.allowInsecureHttp);

  if (config.authRefreshPath) {
    tokens.setRefreshHandler(async (refreshToken) => {
      const refreshed = await client.post<AuthSession>(config.authRefreshPath!, { refreshToken }, { authenticated: false });
      return refreshed.ok ? refreshed.data : null;
    });
  }

  const get = <T>(path: string, opts?: HttpRequestOptions) => client.get<T>(path, opts);
  const post = <T>(path: string, body?: unknown, opts?: HttpRequestOptions) => client.post<T>(path, body, opts);
  const patch = <T>(path: string, body?: unknown, opts?: HttpRequestOptions) => client.patch<T>(path, body, opts);
  const put = <T>(path: string, body?: unknown, opts?: HttpRequestOptions) => client.put<T>(path, body, opts);
  const del = <T>(path: string, opts?: HttpRequestOptions) => client.delete<T>(path, opts);

  const auth: MabizServices["auth"] = {
    register: (input: RegistrationRequest) => post("/auth/register", input, { authenticated: false }),
    async login(input) {
      const result = await post<AuthSession>("/auth/login", input, { authenticated: false });
      if (!isSuccess(result)) return result;
      try {
        await tokens.saveSession(result.data);
        return result;
      } catch {
        return recordFailure("SECURE_SESSION_STORAGE_FAILED", "Secure sign-in could not be completed on this device.");
      }
    },
    async refresh(refreshToken) {
      if (!config.authRefreshPath) return recordFailure("AUTH_REFRESH_NOT_CONFIGURED", "Configure the reviewed refresh route before using API sessions.");
      const token = refreshToken ?? await tokens.getRefreshToken();
      if (!token) return recordFailure("SESSION_EXPIRED", "No refresh credential is available. Sign in again.");
      const result = await post<AuthSession>(config.authRefreshPath, { refreshToken: token }, { authenticated: false });
      if (result.ok) {
        try { await tokens.saveSession(result.data); }
        catch { return recordFailure("SECURE_SESSION_STORAGE_FAILED", "The refreshed session could not be stored securely."); }
      }
      return result;
    },
    async logout() {
      const result = await post<void>("/auth/logout", {});
      await tokens.clearTokens();
      return result;
    },
    requestVerificationCode: (input) => post("/auth/verification-codes", input, { authenticated: false }),
    verifyVerificationCode: (input) => post("/auth/verification-codes/verify", input, { authenticated: false }),
    requestPasswordReset: (input) => post("/auth/password-resets", input, { authenticated: false }),
    completePasswordReset: (input) => post("/auth/password-resets/complete", input, { authenticated: false }),
  };

  const users: MabizServices["users"] = {
    getProfile: (userId) => get(userId === "me" ? "/users/me" : resource("users", userId)),
    updateMyProfile: (input: ProfileUpdate) => patch("/users/me", input),
    listFollowers: async (userId, page) => offsetPageResult(await get<ApiPage<FollowProfileDto>>(resource("users", userId, "followers"), { query: offsetPageQuery(page) }), mapFollowProfile),
    listFollowing: async (userId, page) => offsetPageResult(await get<ApiPage<FollowProfileDto>>(resource("users", userId, "following"), { query: offsetPageQuery(page) }), mapFollowProfile),

/** Gallery accessUrl is a relative API path; images only render once it is qualified with the API origin. */
function resolveApiUrl(config: ApiClientConfig, path?: string | null): string {
  const value = (path ?? "").trim();
  if (!value || /^https?:\/\//i.test(value)) return value;
  const origin = apiOrigin(config);
  return origin ? `${origin}${value.startsWith("/") ? value : `/${value}`}` : value;
}

/** Device file URIs are upload sources, never catalogue references. */
function dropDeviceReferences(value: unknown): unknown {
  if (Array.isArray(value)) return value.filter((entry) => !(typeof entry === "string" && isDeviceFileUri(entry))).map(dropDeviceReferences);
  if (value && typeof value === "object") {
    const entry: Record<string, unknown> = {};
    for (const [key, item] of Object.entries(value as Record<string, unknown>)) {
      if (typeof item === "string" && isDeviceFileUri(item)) continue;
      entry[key] = dropDeviceReferences(item);
    }
    return entry;
  }
  return value;
}

/** Ordered gallery ids; duplicates and blank ids are dropped because the server rejects them outright. */
function catalogGalleryBody(gallery: CatalogGalleryEntry[] | null | undefined): Record<string, unknown> {
  if (!gallery) return {};
  const mediaAssetIds: string[] = [];
  const altTexts: string[] = [];
  for (const item of gallery) {
    const mediaAssetId = item?.mediaAssetId?.trim();
    if (!mediaAssetId || isDeviceFileUri(mediaAssetId) || mediaAssetIds.includes(mediaAssetId)) continue;
    if (mediaAssetIds.length >= MAX_CATALOGUE_GALLERY_IMAGES) break;
    mediaAssetIds.push(mediaAssetId);
    altTexts.push(item.altText?.trim() ?? "");
  }
  return { mediaAssetIds, altTexts };
}

function galleryEntries(gallery: CatalogGalleryImage[] | undefined): CatalogGalleryEntry[] | undefined {
  if (!gallery) return undefined;
  return gallery.map((entry) => ({ mediaAssetId: entry.mediaAssetId, altText: entry.altText }));
}

/** PATCH keeps the stored price unless the client clears it explicitly, so a null price needs `clearPrice`. */
function catalogPriceBody(input: { priceKsh?: number | null; price?: number; priceNote?: string; priceNegotiable?: boolean }, mode: CatalogWriteMode): Record<string, unknown> {
  const priceKsh = input.priceKsh === undefined
    ? (typeof input.price === "number" && Number.isFinite(input.price) ? input.price : null)
    : (typeof input.priceKsh === "number" && Number.isFinite(input.priceKsh) ? input.priceKsh : null);
  const body: Record<string, unknown> = {
    priceKsh,
    priceNote: optionalText(input.priceNote) ?? "",
    priceNegotiable: input.priceNegotiable === true,
  };
  if (mode === "update" && priceKsh === null) body.clearPrice = true;
  return body;
}

/** Products reject lead time, duration, visit type, service areas and any availability enum; the server enforces that split by route. */
function apiProductBody(input: Partial<Product>, mode: CatalogWriteMode): Record<string, unknown> {
  const body: Record<string, unknown> = {
    title: input.name?.trim() ?? "",
    description: input.description?.trim() ?? "",
    category: input.category?.trim() ?? "",
    ...catalogPriceBody(input, mode),
  };
  const sku = optionalText(input.sku);
  if (sku) body.sku = sku.toUpperCase();
  if (typeof input.stockQuantity === "number" && Number.isFinite(input.stockQuantity)) body.stockQuantity = input.stockQuantity;
  if (input.inStock !== undefined) body.inStock = input.inStock;
  const unit = optionalText(input.unit);
  if (unit) body.unit = unit;
  if (input.status === "draft" || input.status === "published") body.status = input.status;
  return dropDeviceReferences({ ...body, ...catalogGalleryBody(galleryEntries(input.gallery)) }) as Record<string, unknown>;
}

/** Services reject SKU, stock and unit; the free-text price label travels as `priceNote`. */
function apiServiceBody(input: Partial<Service>, mode: CatalogWriteMode): Record<string, unknown> {
  const body: Record<string, unknown> = {
    title: input.name?.trim() ?? "",
    description: input.description?.trim() ?? "",
    category: input.category?.trim() ?? "",
    ...catalogPriceBody(input, mode),
    availabilityStatus: input.availabilityStatus ?? catalogAvailabilityStatus(input.availability),
  };
  if (typeof input.durationMinutes === "number" && Number.isFinite(input.durationMinutes)) body.durationMinutes = input.durationMinutes;
  if (typeof input.leadTimeDays === "number" && Number.isFinite(input.leadTimeDays)) body.leadTimeDays = input.leadTimeDays;
  if (input.visitType) body.visitType = input.visitType;
  if (Array.isArray(input.serviceAreas)) body.serviceAreas = input.serviceAreas.map((area) => area.trim()).filter(Boolean).slice(0, 20);
  if (input.status === "draft" || input.status === "published") body.status = input.status;
  return dropDeviceReferences({ ...body, ...catalogGalleryBody(galleryEntries(input.gallery)) }) as Record<string, unknown>;
}

function newInstallationId(): string {
  const cryptoApi = globalThis.crypto as Crypto | undefined;
  if (cryptoApi && typeof cryptoApi.randomUUID === "function") return `install-${cryptoApi.randomUUID()}`;
  return `install-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 22)}`;

/** Requests permission only after an explicit user action; web and simulators are handled without throwing. */
export async function getExpoPushRegistration(): Promise<ExpoPushRegistration> {
  if (Platform.OS !== "ios" && Platform.OS !== "android") throw new Error("Push notifications are available in the iOS and Android apps only.");
  if (!Device.isDevice) throw new Error("Push notification registration requires a physical device.");

  let permission = await Notifications.getPermissionsAsync();
  if (permission.status !== "granted") permission = await Notifications.requestPermissionsAsync();
  if (permission.status !== "granted") throw new Error("Notification permission was not granted on this device.");

  const projectId = Constants.expoConfig?.extra?.eas?.projectId ?? Constants.easConfig?.projectId;
  if (!projectId) throw new Error("Expo Push is not configured for this app build (missing EAS project ID).");

  if (Platform.OS === "android") {
    await Notifications.setNotificationChannelAsync("default", {
      name: "Mabiz notifications",
      importance: Notifications.AndroidImportance.MAX,
      vibrationPattern: [0, 250, 250, 250],
    });
  }

  const token = (await Notifications.getExpoPushTokenAsync({ projectId })).data;
  let deviceIdentifier = await SecureStore.getItemAsync(INSTALLATION_ID_KEY);
  if (!deviceIdentifier) {
    deviceIdentifier = newInstallationId();
    await SecureStore.setItemAsync(INSTALLATION_ID_KEY, deviceIdentifier);
  }

    deviceIdentifier,
    platform: Platform.OS,
    ...(Device.deviceName ? { deviceName: Device.deviceName.slice(0, 100) } : {}),

  };

import type { AuthTokenProvider } from "./http-client";
import type { AuthSession } from "./feature-contracts";

const NATIVE_SESSION_KEY = "Mabiz.api.session.v1";
let browserSession: AuthSession | null = null;

export interface SessionStorage {
  read(): Promise<AuthSession | null>;
  write(session: AuthSession): Promise<void>;
  clear(): Promise<void>;
}

function parseSession(raw: string | null): AuthSession | null {
  if (!raw) return null;
  try {
    const value: unknown = JSON.parse(raw);
    if (!value || typeof value !== "object") return null;
    const session = value as Partial<AuthSession>;
    if (!session.user || typeof session.accessToken !== "string" || typeof session.refreshToken !== "string" || typeof session.expiresAt !== "string") return null;
    return session as AuthSession;
  } catch {
    return null;
  }
}

export class PlatformSessionStorage implements SessionStorage {
  private isBrowser(): boolean {
    return typeof document !== "undefined" && typeof window !== "undefined";
  }

  async read(): Promise<AuthSession | null> {
    if (this.isBrowser()) return browserSession;
    const SecureStore = await import("expo-secure-store");
    const raw = await SecureStore.getItemAsync(NATIVE_SESSION_KEY);
    return parseSession(raw);
  }

  async write(session: AuthSession): Promise<void> {
    if (this.isBrowser()) {
      browserSession = session;
      return;
    }
    const SecureStore = await import("expo-secure-store");
    await SecureStore.setItemAsync(NATIVE_SESSION_KEY, JSON.stringify(session), {
      keychainAccessible: SecureStore.WHEN_UNLOCKED_THIS_DEVICE_ONLY,
    });
  }

  async clear(): Promise<void> {
    if (this.isBrowser()) {
      browserSession = null;
      return;
    }
    const SecureStore = await import("expo-secure-store");
    await SecureStore.deleteItemAsync(NATIVE_SESSION_KEY);
  }
}

export type RefreshSessionHandler = (refreshToken: string) => Promise<AuthSession | null>;

/** Secure bearer-token provider for API mode; no password or demo-session marker is persisted. */
export class SecureSessionTokenProvider implements AuthTokenProvider {
  private refreshHandler?: RefreshSessionHandler;
  private refreshInFlight: Promise<string | null> | null = null;

  constructor(
    private readonly storage: SessionStorage = new PlatformSessionStorage(),
    private readonly sessionExpired?: () => Promise<void> | void,
  ) {}

  setRefreshHandler(handler: RefreshSessionHandler): void {
    this.refreshHandler = handler;
  }

  async getSession(): Promise<AuthSession | null> {
    return this.storage.read();
  }

  async saveSession(session: AuthSession): Promise<void> {
    await this.storage.write(session);
  }

  async getAccessToken(): Promise<string | null> {
    return (await this.storage.read())?.accessToken ?? null;
  }

  async getRefreshToken(): Promise<string | null> {
    return (await this.storage.read())?.refreshToken ?? null;
  }

  refreshAccessToken(): Promise<string | null> {
    if (this.refreshInFlight) return this.refreshInFlight;
    this.refreshInFlight = this.refreshFromStoredSession().finally(() => {
      this.refreshInFlight = null;
    });
    return this.refreshInFlight;
  }

  private async refreshFromStoredSession(): Promise<string | null> {
    const session = await this.storage.read();
    if (!session?.refreshToken || !this.refreshHandler) return null;
    const refreshed = await this.refreshHandler(session.refreshToken);
    if (!refreshed) return null;
    await this.storage.write(refreshed);
    return refreshed.accessToken;
  }

  async clearTokens(): Promise<void> {
    await this.storage.clear();
  }

  async onSessionExpired(): Promise<void> {
    await this.sessionExpired?.();
  }
}

export function isAssistantHistoryEntry(value: unknown): value is AssistantHistoryEntry {
  if (!value || typeof value !== "object") return false;
  const item = value as Record<string, unknown>;
  return typeof item.id === "string" && item.id.length > 0 && typeof item.result === "string" && item.result.trim().length > 0 &&
    typeof item.prompt === "string" && typeof item.purpose === "string" && (CONTENT_PURPOSES as readonly string[]).includes(item.purpose) &&
    typeof item.contentType === "string" && (CONTENT_TYPES as readonly string[]).includes(item.contentType) &&
    typeof item.tone === "string" && (CONTENT_TONES as readonly string[]).includes(item.tone) &&
    typeof item.length === "string" && (CONTENT_LENGTHS as readonly string[]).includes(item.length) &&
    typeof item.createdAt === "string" && Number.isFinite(Date.parse(item.createdAt));
}
export function parseAssistantHistory(raw: string | null): AssistantHistoryEntry[] {
  if (!raw) return [];
  try {
    const parsed: unknown = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed.filter(isAssistantHistoryEntry).slice(0, 50) : [];
  } catch { return []; }
}
export async function readAssistantHistory(): Promise<AssistantHistoryEntry[]> {
  return parseAssistantHistory(await AsyncStorage.getItem(ASSISTANT_HISTORY_KEY));
}
export async function saveAssistantHistory(entry: AssistantHistoryEntry): Promise<AssistantHistoryEntry[]> {
  if (!isAssistantHistoryEntry(entry)) throw new Error("Cannot save an invalid assistant history entry.");
  const current = await readAssistantHistory();
  const next = [entry, ...current.filter((item) => item.id !== entry.id)].slice(0, 50);
  await AsyncStorage.setItem(ASSISTANT_HISTORY_KEY, JSON.stringify(next));
  return next;
}
export async function deleteAssistantHistoryEntry(id: string): Promise<AssistantHistoryEntry[]> {
  const next = (await readAssistantHistory()).filter((item) => item.id !== id);
  await AsyncStorage.setItem(ASSISTANT_HISTORY_KEY, JSON.stringify(next));
  return next;
}
export async function clearAssistantHistory(): Promise<void> {
  await AsyncStorage.removeItem(ASSISTANT_HISTORY_KEY);
}

function page<T>(items: T[], request: PageRequest): PageResponse<T> {
  const limit = Math.min(30, Math.max(1, Math.floor(request.limit || 10)));
  const start = Math.max(0, Number.parseInt(request.cursor ?? "0", 10) || 0);
  const selected = items.slice(start, start + limit);
  const next = start + selected.length;
  return { items: clone(selected), total: items.length, ...(next < items.length ? { nextCursor: String(next) } : {}) };
}
function bySearch<T>(items: T[], request: PageRequest, text: (item: T) => string): T[] {
  const query = request.search?.trim().toLocaleLowerCase();
  return query ? items.filter((item) => text(item).toLocaleLowerCase().includes(query)) : items;
}

        if (!input.participantId && !input.businessId) return fail("CONVERSATION_NOT_FOUND", "Start a conversation before sending a message.");
        const started = await messaging.startConversation({ participantId: input.participantId, businessId: input.businessId, catalogItemId: input.catalogItemId });
        if (!started.ok) return started;
        conversation = conversations.find((item) => item.id === started.data.id);
      }
      if (!conversation) return fail("CONVERSATION_NOT_FOUND", "The local conversation could not be created.");
      const duplicate = conversation.messages.find((message) => message.id === input.clientMessageId);
      if (duplicate) return ok(duplicate);
      const message: ChatMessage = { id: input.clientMessageId || id("message"), senderId: currentUserId, text: input.text, createdAt: now(), status: "sent", ...(input.mediaId ? { image: input.mediaId } : {}) };
      conversation.messages.push(message);
      conversation.lastMessage = input.text || "Shared an attachment";
      conversation.updatedAt = message.createdAt;
      conversation.isArchived = false;
      return ok(message);
    },
    async markDelivered(conversationId) { return conversations.some((item) => item.id === conversationId) ? ok(undefined) : notFound("Conversation"); },
    async markRead(conversationId) { const conversation = conversations.find((item) => item.id === conversationId); if (!conversation) return notFound("Conversation"); conversation.unread = 0; for (const message of conversation.messages) readMessageIds.add(message.id); return ok(undefined); },
    async archive(conversationId) { const conversation = conversations.find((item) => item.id === conversationId); if (!conversation) return notFound("Conversation"); conversation.isArchived = true; return ok(undefined); },
    async block(conversationId) { const conversation = conversations.find((item) => item.id === conversationId); if (!conversation) return notFound("Conversation"); conversation.isBlocked = true; return ok(undefined); },

  };

      const matches: SearchHit[] = [];

      if (allow("business")) for (const business of businesses) if (`${business.name} ${business.category} ${business.subcategory} ${business.town} ${business.neighbourhood}`.toLowerCase().includes(query)) matches.push({ id: business.id, type: "business", title: business.name, subtitle: `${business.category} Â· ${business.town}`, ...(business.cover ? { thumbnailUrl: business.cover } : {}), county: business.county, town: business.town, verified: business.verified });

      if (allow("product")) for (const business of businesses) for (const product of business.products) if (`${product.name} ${product.description} ${product.category}`.toLowerCase().includes(query)) matches.push({ id: product.id, type: "product", title: product.name, subtitle: `${business.name} Â· KSh ${product.price.toLocaleString()}`, ...(product.image ? { thumbnailUrl: product.image } : {}), county: business.county, town: business.town, priceKsh: product.price });
      if (allow("service")) for (const business of businesses) for (const service of business.services) if (`${service.name} ${service.description} ${service.category}`.toLowerCase().includes(query)) matches.push({ id: service.id, type: "service", title: service.name, subtitle: `${business.name} Â· ${service.priceLabel}`, county: business.county, town: business.town });
      if (allow("post") || allow("video")) for (const post of posts) if (`${post.caption} ${post.hashtags.join(" ")}`.toLowerCase().includes(query) && allow(post.video ? "video" : "post")) matches.push({ id: post.id, type: post.video ? "video" : "post", title: post.caption.slice(0, 80), subtitle: post.location, ...(post.media ? { thumbnailUrl: post.media } : {}) });

      }

===== lib/services/service-provider.tsx =====
import { createContext, useContext } from "react";
import { apiConfig, type ApiClientConfig, type ApiMode } from "@/lib/api/config";
import type { MabizServices } from "@/lib/api/feature-contracts";
import { createApiServices } from "@/lib/api/production-services";
import { createMockServices } from "@/lib/services/mock-services";

export interface ServiceRegistryValue {
  mode: ApiMode;
  services: MabizServices;
}

export function createServiceRegistry(mode: ApiMode = apiConfig.mode, config: ApiClientConfig = apiConfig): ServiceRegistryValue {
  if (mode === "api") {
    // API errors remain errors; this factory never falls back to local successes.
    return { mode, services: createApiServices({ config: { ...config, mode } }) };
  }
  return { mode: "mock-local", services: createMockServices() };
}

const defaultRegistry = createServiceRegistry();
const ServicesContext = createContext<ServiceRegistryValue>(defaultRegistry);

export function MabizServicesProvider({
  children,
  registry = defaultRegistry,
}: {
  children: React.ReactNode;
  registry?: ServiceRegistryValue;
}) {
  return <ServicesContext.Provider value={registry}>{children}</ServicesContext.Provider>;
}

export function useMabizServices(): MabizServices {
  return useContext(ServicesContext).services;
}

export function useMabizServiceMode(): ApiMode {
  return useContext(ServicesContext).mode;
}

export const defaultServiceRegistry = defaultRegistry;
